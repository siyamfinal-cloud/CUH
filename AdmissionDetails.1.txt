  id: string;
  name: string;
  classes: string;
  schedule: string;
  admission_fee: string;
  monthly_fee: string;
  term: string;
  start_date: string;
  timing: string;

            <div style={{ display: "grid", gridTemplateColumns: "repeat(3,1fr)", gap: 24 }} className="courses-grid">

                <div
                  key={c.id}
                  style={{
                    background: "white",
                    border: i === 0 ? "2px solid #166534" : "1.5px solid #dcfce7",

                      </div>
                    </div>
                    <Link

                      style={{
                        display: "block",

                        e.currentTarget.style.background = "#14532d";
                        e.currentTarget.style.color = "white";
                      }}
                      onMouseLeave={e => {
                        e.currentTarget.style.background = i === 0 ? "#166534" : "white";

            ].map(s => (
              <div key={s.step} style={{ display: "flex", gap: 16, marginBottom: 20 }}>
                <div style={{
                  width: 36, height: 36, borderRadius: "50%",

                }}>

              key={i}

                border: "1.5px solid #dcfce7",

              )}
            </div>

        @media(max-width:900px){.courses-grid{grid-template-columns:1fr!important;}}
        @media(max-width:640px){.proc-grid{grid-template-columns:1fr!important;}}
      `}</style>
    </div>
