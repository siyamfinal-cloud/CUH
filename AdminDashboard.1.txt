const inp: React.CSSProperties = {
  width: "100%",
  padding: "8px 12px",
  border: "1.5px solid #d1fae5",
  borderRadius: 8,
  fontFamily: "'Hind Siliguri', sans-serif",

  background: "white",
  outline: "none",
  boxSizing: "border-box",
  marginBottom: 12,
};

function Msg({ msg }: { msg: { type: "ok" | "err"; text: string } | null }) {
  if (!msg) return null;
  return (
    <div
      style={{
        padding: "10px 14px",
        borderRadius: 8,
        marginBottom: 16,
        fontSize: "0.84rem",
        background: msg.type === "ok" ? "#dcfce7" : "#fef2f2",
        color: msg.type === "ok" ? "#166534" : "#dc2626",
        border: `1px solid ${msg.type === "ok" ? "#bbf7d0" : "#fecaca"}`,
      }}
    >
      {msg.type === "ok" ? "✅ " : "⚠️ "}
      {msg.text}
    </div>
  );
}

function PhotoUploadInput({
  current,
  onUrl,
  folder,
  bucket,
}: {
  current: string;
  onUrl: (url: string) => void;
  folder: string;
  bucket: string;
}) {
  const [uploading, setUploading] = useState(false);
  const [uploadError, setUploadError] = useState("");

  const handleFile = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;
    setUploading(true);
    setUploadError("");
    const { url, error } = await uploadPhoto(file, folder, bucket);
    if (error) {
      setUploadError("Upload failed: " + error);
    } else if (url) {
      onUrl(url);
    }
    setUploading(false);
    e.target.value = "";
  };

  return (
    <div>
      <label style={{ fontSize: "0.82rem", fontWeight: 600, color: "#374151", display: "block", marginBottom: 4 }}>
        ছবির URL অথবা ফাইল আপলোড
      </label>
      <input
        style={inp}
        type="url"
        value={current}
        onChange={(e) => onUrl(e.target.value)}
        placeholder="https://... অথবা নিচে ফাইল আপলোড করুন"
      />
      <div style={{ display: "flex", alignItems: "center", gap: 10, marginBottom: 12 }}>
        <label
          style={{
            display: "inline-flex",
            alignItems: "center",
            gap: 6,
            padding: "6px 14px",
            background: "#f0fdf4",
            border: "1.5px solid #bbf7d0",
            borderRadius: 7,
            cursor: "pointer",
            fontSize: "0.82rem",
            color: "#166534",
            fontWeight: 600,
          }}
        >
          📁 ফাইল বেছে নিন
          <input type="file" accept="image/*" style={{ display: "none" }} onChange={handleFile} />
        </label>
        {uploading && <span style={{ color: "#6b7280", fontSize: "0.8rem" }}>আপলোড হচ্ছে…</span>}
        {current && !uploading && (
          <img
            src={current}
            alt="preview"
            style={{ width: 40, height: 40, borderRadius: 6, objectFit: "cover", border: "1px solid #dcfce7" }}
          />
        )}
      </div>
      {uploadError && <div style={{ color: "#dc2626", fontSize: "0.78rem", marginBottom: 8 }}>⚠️ {uploadError}</div>}
    </div>
  );
}

// ─── Applications ─────────────────────────────────────────────────────────────

function Applications() {
  const { applications, refreshApplications } = useApp();
  const [search, setSearch] = useState("");
  const [statusFilter, setStatusFilter] = useState("all");
  const [updating, setUpdating] = useState<string | null>(null);
  const [msg, setMsg] = useState<{ type: "ok" | "err"; text: string } | null>(null);

  const filtered = applications.filter((a) => {
    const matchSearch =
      a.studentName.toLowerCase().includes(search.toLowerCase()) ||
      a.refNumber.toLowerCase().includes(search.toLowerCase()) ||
      a.mobile.includes(search);
    const matchStatus = statusFilter === "all" || a.status === statusFilter;
    return matchSearch && matchStatus;
  });

  const updateStatus = async (id: string, status: Application["status"]) => {
    setUpdating(id);
    const { error } = await supabase
      .from("cuh_applications")
      .update({ status })
      .eq("id", id);
    if (error) {
      setMsg({ type: "err", text: "স্ট্যাটাস পরিবর্তন ব্যর্থ: " + error.message });
    } else {
      await refreshApplications();
    }
    setUpdating(null);
  };

  return (
    <div>
      <h2 style={{ fontFamily: "Lora, serif", fontSize: "1.3rem", color: "#14532d", fontWeight: 700, marginBottom: 16 }}>
        ভর্তি আবেদনসমূহ
      </h2>
      <Msg msg={msg} />
      <div style={{ display: "flex", gap: 12, marginBottom: 20, flexWrap: "wrap" }}>
        <input
          className="form-input"
          style={{ maxWidth: 300, marginBottom: 0 }}
          placeholder="নাম, রেফারেন্স, বা মোবাইল"
          value={search}
          onChange={(e) => setSearch(e.target.value)}
        />
        <select
          className="form-input"
          style={{ maxWidth: 200, marginBottom: 0 }}
          value={statusFilter}
          onChange={(e) => setStatusFilter(e.target.value)}
        >
          <option value="all">সব স্ট্যাটাস</option>
          <option value="pending">Pending</option>
          <option value="contacted">Contacted</option>
          <option value="confirmed">Confirmed</option>
          <option value="rejected">Rejected</option>
        </select>
        <button
          onClick={refreshApplications}
          style={{ padding: "8px 16px", border: "1.5px solid #dcfce7", background: "white", borderRadius: 8, cursor: "pointer", fontFamily: "inherit", fontSize: "0.84rem" }}
        >
          🔄 রিফ্রেশ
        </button>
      </div>

      {filtered.length === 0 ? (
        <div style={{ textAlign: "center", color: "#9ca3af", padding: "60px 0" }}>কোনো আবেদন পাওয়া যায়নি।</div>
      ) : (
        <div style={{ overflowX: "auto" }}>
          <table style={{ width: "100%", borderCollapse: "collapse", fontSize: "0.84rem" }}>
            <thead>
              <tr style={{ background: "#f0fdf4", borderBottom: "2px solid #dcfce7" }}>
                {["রেফারেন্স", "নাম", "মোবাইল", "ব্যাচ", "তারিখ", "স্ট্যাটাস", "পরিবর্তন করুন"].map((h) => (
                  <th key={h} style={{ padding: "11px 14px", textAlign: "left", color: "#14532d", fontWeight: 700, whiteSpace: "nowrap" }}>
                    {h}
                  </th>
                ))}
              </tr>
            </thead>
            <tbody>
              {filtered.map((a) => (
                <tr
                  key={a.id}

type TeacherForm = {
  id: string;
  name: string;
  designation: string;
  subject: string;
  qualification: string;
  experience: string;
  bio: string;
  photo_url: string;
  serial_number: string;
};

function TeachersTab() {
  const { teachers, refreshTeachers } = useApp();
  const [editing, setEditing] = useState<TeacherForm | null>(null);
  const [saving, setSaving] = useState(false);
  const [msg, setMsg] = useState<{ type: "ok" | "err"; text: string } | null>(null);

  const blankForm = (): TeacherForm => ({
    id: "",
    name: "",
    designation: "",
    subject: "",
    qualification: "",
    experience: "",
    bio: "",
    photo_url: "",
    serial_number: "",
  });

  const startAdd = () => setEditing(blankForm());
  const startEdit = (t: Teacher) =>
    setEditing({
      id: t.id,
      name: t.name,
      designation: t.designation,
      subject: t.subject,
      qualification: t.qualification,
      experience: t.experience,
      bio: t.bio,
      photo_url: t.photo,
      serial_number: t.serialNumber != null ? String(t.serialNumber) : "",
    });

  const save = async () => {
    if (!editing?.name.trim()) return;
    setSaving(true);
    setMsg(null);

    const payload = {
      name: editing.name,
      designation: editing.designation,
      subject: editing.subject,
      qualification: editing.qualification,
      experience: editing.experience,
      bio: editing.bio,
      photo_url: editing.photo_url,
      image_url: editing.photo_url, // keep both columns in sync for backwards compatibility
      serial_number: editing.serial_number !== "" ? Number(editing.serial_number) : null,
    };

    let error;
    if (editing.id) {
      ({ error } = await supabase.from("cuh_teachers").update(payload).eq("id", editing.id));
    } else {
      ({ error } = await supabase.from("cuh_teachers").insert(payload));
    }

    if (error) {
      setMsg({ type: "err", text: "সংরক্ষণ ব্যর্থ: " + error.message });
    } else {
      setMsg({ type: "ok", text: "শিক্ষকের তথ্য সংরক্ষিত হয়েছে।" });
      await refreshTeachers();
      setEditing(null);
    }
    setSaving(false);
  };

  const del = async (id: string, name: string) => {
    if (!confirm(`"${name}" কে মুছে ফেলতে চান?`)) return;
    const { error } = await supabase.from("cuh_teachers").delete().eq("id", id);
    if (error) {
      setMsg({ type: "err", text: "মুছতে ব্যর্থ: " + error.message });
    } else {
      setMsg({ type: "ok", text: "শিক্ষক মুছে ফেলা হয়েছে।" });
      await refreshTeachers();
    }
  };

  const set = (k: keyof TeacherForm) => (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) =>
    setEditing((f) => f && { ...f, [k]: e.target.value });

  if (editing !== null) {
    return (
      <div>
        <h2 style={{ fontFamily: "Lora, serif", fontSize: "1.3rem", color: "#14532d", fontWeight: 700, marginBottom: 20 }}>
          {editing.id ? "শিক্ষক সম্পাদনা" : "নতুন শিক্ষক যোগ করুন"}
        </h2>
        <Msg msg={msg} />
        <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: "0 20px" }} className="form-grid">
          {[
            ["নাম *", "name"],
            ["পদবী", "designation"],
            ["বিষয়", "subject"],
            ["শিক্ষাগত যোগ্যতা", "qualification"],
            ["অভিজ্ঞতা", "experience"],
          ].map(([label, key]) => (
            <div key={key}>
              <label style={{ fontSize: "0.82rem", fontWeight: 600, color: "#374151", display: "block", marginBottom: 4 }}>{label}</label>
              <input style={inp} value={(editing as Record<string, string>)[key]} onChange={set(key as keyof TeacherForm)} />
            </div>
          ))}
          <div>
            <label style={{ fontSize: "0.82rem", fontWeight: 600, color: "#374151", display: "block", marginBottom: 4 }}>ক্রমিক নম্বর</label>
            <input
              style={inp}
              type="number"
              placeholder="যেমন: 1"
              value={editing.serial_number}
              onChange={(e) => setEditing((f) => f && { ...f, serial_number: e.target.value })}
            />
          </div>
        </div>

        <label style={{ fontSize: "0.82rem", fontWeight: 600, color: "#374151", display: "block", marginBottom: 4 }}>
          সংক্ষিপ্ত পরিচিতি
        </label>
        <textarea
          style={{ ...inp, height: 80, resize: "vertical" }}
          value={editing.bio}
          onChange={set("bio")}
        />

        <PhotoUploadInput
          current={editing.photo_url}
          onUrl={(url) => setEditing((f) => f && { ...f, photo_url: url })}
          folder="teachers"
          bucket={TEACHER_BUCKET}
        />

        <div style={{ display: "flex", gap: 12, marginTop: 8 }}>
          <button
            onClick={save}
            disabled={saving}
            className="btn-primary"
            style={{ opacity: saving ? 0.7 : 1 }}
          >
            {saving ? "সংরক্ষণ হচ্ছে…" : "সংরক্ষণ করুন"}
          </button>
          <button
            onClick={() => { setEditing(null); setMsg(null); }}
            style={{ padding: "10px 20px", border: "1.5px solid #dcfce7", background: "white", borderRadius: 8, cursor: "pointer", fontFamily: "inherit" }}
          >
            বাতিল
          </button>
        </div>
        <style>{`@media(max-width:640px){.form-grid{grid-template-columns:1fr!important;}}`}</style>
      </div>
    );
  }

  return (
    <div>
      <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", marginBottom: 20 }}>
        <h2 style={{ fontFamily: "Lora, serif", fontSize: "1.3rem", color: "#14532d", fontWeight: 700 }}>শিক্ষকমণ্ডলী</h2>
        <button onClick={startAdd} className="btn-primary">+ নতুন শিক্ষক</button>
      </div>
      <Msg msg={msg} />
      {teachers.length === 0 && (
        <div style={{ textAlign: "center", color: "#9ca3af", padding: "40px 0" }}>কোনো শিক্ষক নেই। যোগ করুন।</div>
      )}
      <div style={{ display: "flex", flexDirection: "column", gap: 12 }}>
        {teachers.map((t) => (
          <div key={t.id} style={{ background: "white", border: "1.5px solid #dcfce7", borderRadius: 12, padding: "14px 18px", display: "flex", alignItems: "center", gap: 16 }}>
            <div style={{
              width: 32, height: 32, borderRadius: 8, flexShrink: 0,
              background: t.serialNumber != null ? "#dcfce7" : "#f3f4f6",
              border: "1.5px solid " + (t.serialNumber != null ? "#bbf7d0" : "#e5e7eb"),
              display: "flex", alignItems: "center", justifyContent: "center",
              fontWeight: 700, fontSize: "0.8rem",
              color: t.serialNumber != null ? "#166534" : "#9ca3af",
            }}>
              {t.serialNumber != null ? t.serialNumber : "—"}
            </div>
            <img
              src={t.photo || "https://placehold.co/56x56/dcfce7/166534?text=T"}
              alt={t.name}
              style={{ width: 52, height: 52, borderRadius: "50%", objectFit: "cover", flexShrink: 0 }}
            />
            <div style={{ flex: 1 }}>
              <div style={{ fontWeight: 700, color: "#14532d" }}>{t.name}</div>
              <div style={{ color: "#166534", fontSize: "0.82rem" }}>
                {t.subject} — {t.designation}
              </div>
            </div>
            <div style={{ display: "flex", gap: 8 }}>
              <button
                onClick={() => startEdit(t)}
                style={{ padding: "6px 14px", border: "1px solid #dcfce7", background: "white", borderRadius: 6, cursor: "pointer", fontSize: "0.8rem", fontFamily: "inherit" }}

  name: string;
  classes: string;
  schedule: string;
  admission_fee: string;
  monthly_fee: string;
  term: string;
  start_date: string;
  timing: string;
};

const COURSES_SETUP_SQL = `-- Run once in: Supabase Dashboard → SQL Editor → New query

CREATE EXTENSION IF NOT EXISTS "pgcrypto";

CREATE TABLE IF NOT EXISTS public.cuh_courses (
  id            UUID        PRIMARY KEY DEFAULT gen_random_uuid(),
  name          TEXT        NOT NULL DEFAULT '',
  classes       TEXT        NOT NULL DEFAULT '',
  schedule      TEXT        NOT NULL DEFAULT '',
  admission_fee TEXT        NOT NULL DEFAULT '',
  monthly_fee   TEXT        NOT NULL DEFAULT '',
  term          TEXT        NOT NULL DEFAULT '',
  start_date    TEXT        NOT NULL DEFAULT '',
  timing        TEXT        NOT NULL DEFAULT '',
  created_at    TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at    TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

ALTER TABLE public.cuh_courses ENABLE ROW LEVEL SECURITY;

DO $do$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE tablename='cuh_courses' AND policyname='public_read_courses') THEN
    EXECUTE 'CREATE POLICY "public_read_courses" ON public.cuh_courses FOR SELECT USING (TRUE)';
  END IF;
  IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE tablename='cuh_courses' AND policyname='auth_all_courses') THEN
    EXECUTE 'CREATE POLICY "auth_all_courses" ON public.cuh_courses FOR ALL USING (auth.uid() IS NOT NULL)';
  END IF;
END $do$;

NOTIFY pgrst, 'reload schema';`;

function CoursesTab() {
  const { courses, refreshCourses } = useApp();
  const [editing, setEditing] = useState<CourseForm | null>(null);
  const [saving, setSaving] = useState(false);
  const [msg, setMsg] = useState<{ type: "ok" | "err"; text: string } | null>(null);
  const [tableError, setTableError] = useState<string | null>(null);
  const [copied, setCopied] = useState(false);

  // Probe the table on mount to detect missing-table error immediately.
  useEffect(() => {
    supabase
      .from("cuh_courses")
      .select("id", { count: "exact", head: true })
      .then(({ error }) => {
        if (error && (error.message.includes("schema cache") || error.message.includes("does not exist"))) {
          setTableError(error.message);
        } else {
          setTableError(null);
        }
      });
  }, []);

  const handleCopySQL = () => {
    navigator.clipboard.writeText(COURSES_SETUP_SQL).then(() => {
      setCopied(true);
      setTimeout(() => setCopied(false), 2500);
    });
  };

  const handleRetry = async () => {
    setTableError(null);
    await refreshCourses();
    // Re-probe
    const { error } = await supabase.from("cuh_courses").select("id", { count: "exact", head: true });
    if (error && (error.message.includes("schema cache") || error.message.includes("does not exist"))) {
      setTableError(error.message);
    }
  };

  // Show setup panel when table doesn't exist
  if (tableError) {
    return (
      <div>
        <h2 style={{ fontFamily: "Lora, serif", fontSize: "1.3rem", color: "#14532d", fontWeight: 700, marginBottom: 20 }}>কোর্সসমূহ</h2>
        <div style={{ background: "#fff7ed", border: "1.5px solid #fed7aa", borderRadius: 12, padding: "20px 24px", marginBottom: 20 }}>
          <div style={{ fontWeight: 700, color: "#9a3412", marginBottom: 8 }}>⚠️ cuh_courses টেবিল তৈরি হয়নি</div>
          <p style={{ color: "#c2410c", fontSize: "0.88rem", lineHeight: 1.7, margin: "0 0 16px" }}>
            নিচের SQL টি <strong>Supabase Dashboard → SQL Editor → New query</strong>-তে paste করে Run করুন। তারপর "পুনরায় চেষ্টা করুন" বাটনে ক্লিক করুন।
          </p>
          <div style={{ position: "relative" }}>
            <pre style={{
              background: "#1e293b", color: "#e2e8f0", fontSize: "0.75rem", lineHeight: 1.7,
              padding: "16px", borderRadius: 8, overflowX: "auto", margin: 0,
              fontFamily: "monospace", whiteSpace: "pre-wrap", wordBreak: "break-all",
            }}>
              {COURSES_SETUP_SQL}
            </pre>
            <button
              onClick={handleCopySQL}
              style={{
                position: "absolute", top: 8, right: 8,
                padding: "5px 12px", borderRadius: 6, border: "none", cursor: "pointer",
                background: copied ? "#166534" : "#475569", color: "white",
                fontSize: "0.75rem", fontWeight: 600, fontFamily: "inherit",
              }}
            >
              {copied ? "✓ Copied!" : "Copy SQL"}
            </button>
          </div>
          <div style={{ marginTop: 16 }}>
            <button onClick={handleRetry} className="btn-primary">🔄 পুনরায় চেষ্টা করুন</button>
          </div>
        </div>
      </div>
    );
  }

  const blankForm = (): CourseForm => ({ id: "", name: "", classes: "", schedule: "", admission_fee: "", monthly_fee: "", term: "", start_date: "", timing: "" });
  const toForm = (c: Course): CourseForm => ({ id: c.id, name: c.name, classes: c.classes, schedule: c.schedule, admission_fee: c.admissionFee, monthly_fee: c.monthlyFee, term: c.term, start_date: c.startDate, timing: c.timing });

  const save = async () => {

    } else {
      ({ error } = await supabase.from("cuh_courses").insert(payload));

    if (error) {
      setMsg({ type: "err", text: "সংরক্ষণ ব্যর্থ: " + error.message });
    } else {
      setMsg({ type: "ok", text: "কোর্স সংরক্ষিত হয়েছে।" });

DO $do$

-- Storage bucket policies for gallery-images

DO $do$

      </div>
    </div>
  );

export default function AdminDashboard() {
  const { isAdminLoggedIn, adminLogout, session } = useApp();
  const navigate = useNavigate();
  const [activeTab, setActiveTab] = useState<Tab>("overview");

  useEffect(() => {
    if (!isAdminLoggedIn) navigate("/admin");
  }, [isAdminLoggedIn, navigate]);

  const handleLogout = async () => {
    await adminLogout();
    navigate("/admin");
  };

  const renderTab = () => {
    switch (activeTab) {
      case "overview": return <Overview />;
      case "applications": return <Applications />;
      case "teachers": return <TeachersTab />;
      case "courses": return <CoursesTab />;
      case "notices": return <NoticesTab />;
      case "stories": return <StoriesTab />;
      case "gallery": return <GalleryTab />;
      case "stats": return <StatsTab />;
      case "settings": return <SettingsTab />;
    }
  };

  return (
    <div style={{ display: "flex", minHeight: "100vh", fontFamily: "'Hind Siliguri', sans-serif", background: "#f8fafb" }}>
      {/* Sidebar */}
      <div className="admin-sidebar" style={{ display: "flex", flexDirection: "column", position: "sticky", top: 0, height: "100vh", overflowY: "auto" }}>
        <div style={{ padding: "20px 16px 12px", borderBottom: "1px solid rgba(255,255,255,0.1)" }}>
          <div style={{ display: "flex", alignItems: "center", gap: 10, marginBottom: 4 }}>
            <div style={{ width: 36, height: 36, borderRadius: 8, background: "rgba(255,255,255,0.15)", display: "flex", alignItems: "center", justifyContent: "center", color: "white", fontWeight: 700, fontFamily: "Lora, serif" }}>C</div>
            <div>
              <div style={{ color: "white", fontWeight: 700, fontSize: "0.85rem" }}>Admin Panel</div>

            </div>

          )}

          {tabs.map((t) => (
            <button
              key={t.id}
              onClick={() => setActiveTab(t.id)}
              className={`admin-nav-item ${activeTab === t.id ? "active" : ""}`}
              style={{ width: "100%", border: "none", background: "none", cursor: "pointer", textAlign: "left", fontFamily: "inherit" }}

            style={{ width: "100%", padding: "10px", background: "rgba(255,255,255,0.1)", color: "#bbf7d0", border: "1px solid rgba(255,255,255,0.15)", borderRadius: 8, cursor: "pointer", fontFamily: "inherit", fontSize: "0.85rem", fontWeight: 500 }}

        </div>
