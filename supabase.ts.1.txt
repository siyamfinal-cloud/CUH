import { createClient } from "@supabase/supabase-js";
import { projectId, publicAnonKey } from "../../utils/supabase/info";

// In Figma Make's preview iframe, localStorage may be sandboxed (throws or
// silently drops writes). Verify both write and read-back before trusting it;
// fall back to in-memory storage so createClient never crashes.
function safeStorage(): Pick<Storage, "getItem" | "setItem" | "removeItem"> {
  try {
    const testKey = "__cuh_storage_test__";
    const testVal = "cuh_ok";
    localStorage.setItem(testKey, testVal);
    const readBack = localStorage.getItem(testKey);
    localStorage.removeItem(testKey);
    if (readBack === testVal) return localStorage;
  } catch {
    // fall through
  }
  // in-memory fallback — singleton so module re-imports share the same store
  const mem: Record<string, string> = {};
  return {
    getItem: (k) => mem[k] ?? null,
    setItem: (k, v) => { mem[k] = v; },
    removeItem: (k) => { delete mem[k]; },
  };
}

export const supabase = createClient(
  `https://${projectId}.supabase.co`,
  publicAnonKey,
  {
    auth: {
      persistSession: true,
      autoRefreshToken: true,
      // Unique key prevents collision with Figma Make's own Supabase client
      // (which also connects to the same project), eliminating the
      // "Multiple GoTrueClient instances" warning and auth state interference.
      storageKey: "cuh-supabase-auth",
      storage: safeStorage(),
    },
  }
);

// Upload a file to Supabase Storage and return its public URL.
// Works in both normal and Incognito sessions.
export async function uploadPhoto(
  file: File,
  folder: string,
  bucket: string
): Promise<{ url: string | null; error: string | null }> {

  // ── 1. Verify authenticated session ───────────────────────────────────────
  const { data: { session }, error: sessionError } = await supabase.auth.getSession();

  console.info(
    "[uploadPhoto] project:", projectId,
    "| bucket:", bucket,
    "| user:", session?.user?.id ?? "none",
    "| token-prefix:", session?.access_token?.slice(0, 20) ?? "none",
    "| session-error:", sessionError?.message ?? "none",
  );

  if (sessionError || !session) {
    return {
      url: null,
      error: "ফাইল আপলোড করতে লগইন করুন। অ্যাডমিন প্যানেলে প্রবেশ করুন এবং পুনরায় চেষ্টা করুন।",
    };
  }

  // ── 2. Refresh session to guarantee a fresh JWT is set on the client ───────
  // supabase-js propagates the refreshed token to Storage automatically via
  // the shared GoTrueClient reference — no need to pass the token manually.
  const { data: { session: refreshed }, error: refreshError } =
    await supabase.auth.refreshSession();

  if (refreshError) {
    console.warn("[uploadPhoto] session refresh warning:", refreshError.message);
    // Do NOT abort — the existing session's access_token may still be valid.
  } else {
    console.info("[uploadPhoto] session refreshed | user:", refreshed?.user?.id ?? "none");
  }

  // ── 3. Upload ─────────────────────────────────────────────────────────────
  // NOTE: getBucket() is a service_role-only management API and will always
  // return "Bucket not found" for regular authenticated users — do not call it.
  const ext = file.name.split(".").pop() ?? "jpg";
  const path = `${folder}/${Date.now()}-${Math.random().toString(36).slice(2)}.${ext}`;

  const { data, error: uploadError } = await supabase.storage
    .from(bucket)
    .upload(path, file, {
      cacheControl: "3600",
      upsert: false,
      contentType: file.type || "application/octet-stream",
    });

  if (uploadError) {
    console.error("[uploadPhoto] upload error:", uploadError.message, "| status:", (uploadError as { statusCode?: string }).statusCode);
    return { url: null, error: uploadError.message };
  }

  const { data: { publicUrl } } = supabase.storage
    .from(bucket)
    .getPublicUrl(data.path);

  console.info("[uploadPhoto] success | publicUrl:", publicUrl);
  return { url: publicUrl, error: null };
}

// Row → frontend type transformers
export function rowToTeacher(r: Record<string, unknown>) {
  // Prefer photo_url; fall back to image_url (older column name); then placeholder.
  const photoUrl: string =
    (r.photo_url ? String(r.photo_url).trim() : "") ||
    (r.image_url ? String(r.image_url).trim() : "") ||
    "https://placehold.co/200x200/dcfce7/166534?text=T";
  return {
    id: String(r.id),
    name: String(r.name ?? ""),
    designation: String(r.designation ?? ""),
    subject: String(r.subject ?? ""),
    qualification: String(r.qualification ?? ""),
    experience: String(r.experience ?? ""),
    bio: String(r.bio ?? ""),
    photo: photoUrl,
    serialNumber: r.serial_number != null ? Number(r.serial_number) : null,
  };
}
